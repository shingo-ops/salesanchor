     1	#!/usr/bin/env python3
     2	"""migration の値の書き込み検査（ADR-155: マイグレーションは構造変更のみ）。
     3	
     4	保護対象の表（scripts/migration-guard/protected-tables.txt）への INSERT / UPDATE / DELETE と、
     5	データの参照（SELECT 等）を見つける。標準ライブラリだけで動く。
     6	
     7	使い方:
     8	  python3 scripts/check-migration-value-writes.py diff BASE HEAD [--check 7|8|all]
     9	      PR の差分を検査する（チェック 7・8）。新規ファイルは全体、変更されたファイルは「追加された行」だけを見る。
    10	      neutralize（値の書き込みを外し、コメントと NOTICE を足す）編集は通る。
    11	  python3 scripts/check-migration-value-writes.py repo
    12	      scripts/run_all_migrations.sh に登録された全ファイルを走査し、値の書き込みを持つファイルが
    13	      value-write-allowlist.tsv にあることを確かめる（許可一覧に古い行が残っていても失敗する）。
    14	"""
    15	from __future__ import annotations
    16	
    17	import argparse
    18	import re
    19	import subprocess
    20	import sys
    21	from pathlib import Path
    22	
    23	ROOT = Path(__file__).resolve().parent.parent
    24	GUARD_DIR = ROOT / "scripts" / "migration-guard"
    25	TABLES_FILE = GUARD_DIR / "protected-tables.txt"
    26	ALLOWLIST_FILE = GUARD_DIR / "value-write-allowlist.tsv"
    27	RUNNER = ROOT / "scripts" / "run_all_migrations.sh"
    28	
    29	MIGRATION_PATH = re.compile(r"^migrations/[0-9].*\.sql$")
    30	# チェック 7 と同じ正規表現。違いは 2 つ: (1) 動的なスキーマ名（%I. / %s.）も許す（tenant ループの
    31	# EXECUTE format で書かれた値の書き込みを見逃さないため）、(2) 表名の後ろに語境界（products が
    32	# products_xxx に当たらない）。
    33	SCHEMA_PREFIX = r"(?:public\.|tenant_[0-9]+\.|%I\.|%s\.)?"
    34	# チェック 8 の許可パターン（構造変更・存在確認・NOTICE の行は、表名が出ても許す）
    35	ALLOW_PATTERN = re.compile(
    36	    r"CREATE\s+TABLE|ALTER\s+TABLE|DROP\s+TABLE|CREATE\s+(UNIQUE\s+)?INDEX|DROP\s+INDEX|DROP\s+TRIGGER|DROP\s+VIEW"
    37	    r"|CREATE\s+(OR\s+REPLACE\s+)?TRIGGER|COMMENT\s+ON|ADD\s+COLUMN|DROP\s+COLUMN|ALTER\s+COLUMN|REFERENCES"
    38	    r"|information_schema|pg_namespace|pg_class|pg_trigger|to_regclass|RAISE\s+NOTICE"
    39	    r"|CREATE\s+(OR\s+REPLACE\s+)?VIEW|CREATE\s+(OR\s+REPLACE\s+)?FUNCTION|RENAME\s+TO|table_name|table_schema"
    40	    r"|PRIMARY\s+KEY|CREATE\s+POLICY|DROP\s+POLICY|ALTER\s+POLICY",
    41	    re.IGNORECASE,
    42	)
    43	
    44	
    45	def load_tables(path: Path = TABLES_FILE) -> list[str]:
    46	    names = []
    47	    for raw in path.read_text(encoding="utf-8").splitlines():
    48	        line = raw.strip()
    49	        if line and not line.startswith("#"):
    50	            names.append(line)
    51	    if not names:
    52	        raise SystemExit(f"保護対象の表の一覧が空です: {path}")
    53	    return names
    54	
    55	
    56	def build_regexes(tables: list[str]) -> dict[str, re.Pattern[str]]:
    57	    alt = "|".join(re.escape(t) for t in sorted(tables, key=len, reverse=True))
    58	    table = rf"({alt})\b"
    59	    return {
    60	        "INSERT": re.compile(rf"INSERT\s+INTO\s+{SCHEMA_PREFIX}{table}", re.IGNORECASE),
    61	        "UPDATE": re.compile(rf"UPDATE\s+{SCHEMA_PREFIX}{table}", re.IGNORECASE),
    62	        "DELETE": re.compile(rf"DELETE\s+FROM\s+{SCHEMA_PREFIX}{table}", re.IGNORECASE),
    63	    }
    64	
    65	
    66	def build_reference_regex(tables: list[str]) -> re.Pattern[str]:
    67	    alt = "|".join(re.escape(t) for t in sorted(tables, key=len, reverse=True))
    68	    return re.compile(rf"\b(?:{alt})\b", re.IGNORECASE)
    69	
    70	
    71	def is_comment(line: str) -> bool:
    72	    return line.lstrip().startswith("--")
    73	
    74	
    75	def normalize(line: str) -> str:
    76	    return " ".join(line.split())
    77	
    78	
    79	def value_writes(lines: list[str], regexes: dict[str, re.Pattern[str]]) -> list[tuple[int, str, str, str]]:
    80	    """(行番号, 種類, 表名, 行) のリスト。コメント行は見ない。"""
    81	    found = []
    82	    for number, line in enumerate(lines, 1):
    83	        if is_comment(line):
    84	            continue
    85	        for kind, regex in regexes.items():
    86	            match = regex.search(line)
    87	            if match:
    88	                found.append((number, kind, match.group(1).lower(), line.strip()))
    89	    return found
    90	
    91	
    92	def git(*args: str) -> str:
    93	    return subprocess.run(["git", *args], cwd=ROOT, capture_output=True, text=True, check=True).stdout
    94	
    95	
    96	# ---------------------------------------------------------------- diff モード（チェック 7・8）
    97	def changed_migrations(base: str, head: str) -> list[tuple[str, list[str], str]]:
    98	    """(状態 A/M/R, 差分を取るパス, 新しいパス)。migrations/ の .sql だけ。"""
    99	    out = git("diff", "-M", "--name-status", "--diff-filter=AMR", base, head, "--", "migrations")
   100	    files = []
   101	    for row in out.splitlines():
   102	        parts = row.split("\t")
   103	        status = parts[0][0]
   104	        new = parts[-1]
   105	        if not MIGRATION_PATH.match(new):
   106	            continue
   107	        paths = [parts[1], parts[2]] if status == "R" else [new]
   108	        files.append((status, paths, new))
   109	    return files
   110	
   111	
   112	def added_non_comment_lines(base: str, head: str, paths: list[str]) -> list[str]:
   113	    """差分で追加された行（コメント行を除く）。同じ文面が削除された行にもあれば（字下げの変更・移動）除く。"""
   114	    diff = git("diff", "-M", base, head, "--", *paths)
   115	    added = [ln[1:] for ln in diff.splitlines() if ln.startswith("+") and not ln.startswith("+++")]
   116	    removed = {normalize(ln[1:]) for ln in diff.splitlines() if ln.startswith("-") and not ln.startswith("---")}
   117	    return [ln for ln in added if not is_comment(ln) and normalize(ln) not in removed]
   118	
   119	
   120	def run_diff(base: str, head: str, check: str) -> int:
   121	    tables = load_tables()
   122	    regexes = build_regexes(tables)
   123	    reference = build_reference_regex(tables)
   124	    files = changed_migrations(base, head)
   125	    failed7: list[str] = []
   126	    failed8: list[str] = []
   127	    for status, paths, new in files:
   128	        lines = (
   129	            git("show", f"{head}:{new}").splitlines() if status == "A" else added_non_comment_lines(base, head, paths)
   130	        )
   131	        if status == "A":  # 新規ファイルは全体を見る（コメント行は除く）
   132	            lines = [ln for ln in lines if not is_comment(ln)]
   133	        hits = value_writes(lines, regexes)
   134	        if hits and check in ("7", "all"):
   135	            failed7.append(f"--- {Path(new).name} ---")
   136	            failed7.extend(f"  {kind} {table}: {line}" for _, kind, table, line in hits)
   137	        if check in ("8", "all"):
   138	            bad = [ln.strip() for ln in lines if reference.search(ln) and not ALLOW_PATTERN.search(ln)]
   139	            if bad:
   140	                failed8.append(f"--- {Path(new).name} ---")
   141	                failed8.extend(f"  {line}" for line in bad)
   142	    if not files:
   143	        print("✅ 追加・変更された migrations/*.sql なし — チェック 7・8 スキップ")
   144	        return 0
   145	    status_code = 0
   146	    if check in ("7", "all"):
   147	        if failed7:
   148	            status_code = 1
   149	            print("❌ MIGRATION GUARD FAILED (チェック7)")
   150	            print(f"共用マスタテーブル（{len(tables)} 表: scripts/migration-guard/protected-tables.txt）への")
   151	            print("INSERT / UPDATE / DELETE が、追加された行に検出されました（ADR-155）。")
   152	            print("\n".join(failed7))
   153	            print("マイグレーションは構造変更のみ。値はアプリ画面か CSV で変える。参考: ADR-155-product-master-ssot-csv-app.md")
   154	        else:
   155	            print(f"✅ 共用マスタテーブルへのデータ操作なし（{len(files)} ファイル）— チェック7 通過")
   156	    if check in ("8", "all"):
   157	        if failed8:
   158	            status_code = 1
   159	            print("❌ MIGRATION GUARD FAILED (チェック8)")
   160	            print("共用マスタテーブルのデータ参照（構造変更・存在確認・NOTICE 以外）が、追加された行に検出されました（ADR-155）。")
   161	            print("\n".join(failed8))
   162	        else:
   163	            print(f"✅ 共用マスタテーブルへのデータ参照なし（{len(files)} ファイル）— チェック8 通過")
   164	    return status_code
   165	
   166	
   167	# ---------------------------------------------------------------- repo モード（全件走査）
   168	def registered_files() -> list[str]:
   169	    files = []
   170	    for line in RUNNER.read_text(encoding="utf-8").splitlines():
   171	        if line.startswith("run_sql "):
   172	            files.append(line.split()[1])
   173	    return list(dict.fromkeys(files))
   174	
   175	
   176	def load_allowlist(path: Path = ALLOWLIST_FILE) -> dict[str, tuple[set[str], str]]:
   177	    rows: dict[str, tuple[set[str], str]] = {}
   178	    for number, raw in enumerate(path.read_text(encoding="utf-8").splitlines(), 1):
   179	        if not raw.strip() or raw.lstrip().startswith("#"):
   180	            continue
   181	        cols = raw.split("\t")
   182	        if len(cols) < 4 or not all(c.strip() for c in cols[:4]):
   183	            raise SystemExit(f"{path}:{number}: 列は file<TAB>tables<TAB>reason<TAB>decision_ref の 4 つが必要です")
   184	        rows[cols[0].strip()] = ({t.strip().lower() for t in cols[1].split(",") if t.strip()}, cols[2].strip())
   185	    return rows
   186	
   187	
   188	def scan_registered() -> dict[str, set[str]]:
   189	    regexes = build_regexes(load_tables())
   190	    flagged: dict[str, set[str]] = {}
   191	    for path in registered_files():
   192	        file = ROOT / path
   193	        if not file.is_file():
   194	            continue
   195	        hits = value_writes(file.read_text(encoding="utf-8").splitlines(), regexes)
   196	        if hits:
   197	            flagged[path] = {table for _, _, table, _ in hits}
   198	    return flagged
   199	
   200	
   201	def run_repo() -> int:
   202	    registered = set(registered_files())
   203	    flagged = scan_registered()
   204	    allow = load_allowlist()
   205	    problems = []
   206	    for path, tables in sorted(flagged.items()):
   207	        if path not in allow:
   208	            problems.append(f"許可一覧にない値の書き込み: {path}（{', '.join(sorted(tables))}）")
   209	        elif allow[path][0] != tables:
   210	            problems.append(
   211	                f"許可一覧の tables が実際と違う: {path}（一覧 {sorted(allow[path][0])} / 実際 {sorted(tables)}）"
   212	            )
   213	    for path in sorted(allow):
   214	        if path not in registered:
   215	            problems.append(f"許可一覧の行が、登録されていない（または無い）ファイルを指している: {path}")
   216	        elif path not in flagged:
   217	            problems.append(f"許可一覧の古い行（このファイルに値の書き込みはもう無い。行を消す）: {path}")
   218	    if problems:
   219	        print("❌ MIGRATION GUARD FAILED (全件走査)")
   220	        print("\n".join(f"  - {p}" for p in problems))
   221	        print("直し方: 値の書き込みは migration から外す（neutralize）か、残すものは理由つきで")
   222	        print("scripts/migration-guard/value-write-allowlist.tsv に書く。外したファイルの行は同じ PR で消す。")
   223	        return 1
   224	    print(f"✅ 登録された {len(registered)} ファイルを走査: 値の書き込みを持つ {len(flagged)} ファイル = 許可一覧 {len(allow)} 行 — 全件走査 通過")
   225	    return 0
   226	
   227	
   228	def main() -> int:
   229	    parser = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
   230	    sub = parser.add_subparsers(dest="mode", required=True)
   231	    diff = sub.add_parser("diff")
   232	    diff.add_argument("base")
   233	    diff.add_argument("head")
   234	    diff.add_argument("--check", choices=["7", "8", "all"], default="all")
   235	    sub.add_parser("repo")
   236	    args = parser.parse_args()
   237	    if args.mode == "diff":
   238	        return run_diff(args.base, args.head, args.check)
   239	    return run_repo()
   240	
   241	
   242	if __name__ == "__main__":
   243	    sys.exit(main())
