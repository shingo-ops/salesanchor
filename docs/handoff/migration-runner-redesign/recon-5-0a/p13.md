MIGS = set(os.listdir('migrations'))
def py_sql_names(path):
    """P13 rule: strings ending .sql in the script body that exist in migrations/, order of appearance, unique."""
    body = read(path)
    names = []
    for m in re.finditer(r'[A-Za-z0-9_./\-]*[A-Za-z0-9_\-]\.sql', body):
        base = os.path.basename(m.group(0))
        if base in MIGS and base not in names:
            names.append(base)
    return names
