```python
import json, subprocess, sys
runs = json.load(open('/tmp/recon/p09_runs.json'))
for r in runs[:30]:
    rid = r['databaseId']
    out = subprocess.run(['gh', 'run', 'view', str(rid), '--repo', 'shingo-ops/salesanchor', '--json', 'jobs,headSha'], capture_output=True, text=True)
    if out.returncode != 0:
        print(rid, 'ERR', out.stderr[:200]); continue
    d = json.loads(out.stdout)
    for j in d['jobs']:
        mig = [s for s in j.get('steps', []) if 'igrat' in s['name'] and s['conclusion'] == 'success']
        print(rid, r['createdAt'], d.get('headSha', '')[:9], j['databaseId'], j['name'], j['conclusion'], '| migration steps:', [(s['name'], s['startedAt'], s['completedAt']) for s in mig])
```
