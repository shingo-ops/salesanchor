## 台本の全文（/tmp/recon/common.py）
```python
import re, os, sys
ROOT = '/root/worktrees/salesanchor/release-migration-skip-applied-5-0'
os.chdir(ROOT)

def registered():
    out = []
    for i, line in enumerate(open('scripts/run_all_migrations.sh', encoding='utf-8'), 1):
        m = re.match(r'^run_(sql|py)\s+(\S+)', line)
        if m:
            out.append((i, m.group(1), m.group(2)))
    return out

def strip_sql_comments(s):
    s = re.sub(r'/\*.*?\*/', lambda m: '\n' * m.group(0).count('\n'), s, flags=re.S)
    lines = []
    for ln in s.split('\n'):
        k = ln.find('--')
        lines.append(ln if k < 0 else ln[:k])
    return '\n'.join(lines)

def read(p):
    return open(p, encoding='utf-8', errors='replace').read()

MIGS = set(os.listdir('migrations'))

def py_sql_names(path):
    """P13 rule: strings ending .sql in the script body that exist in migrations/, order of appearance, unique."""
    body = read(path)
    names = []
    for m in re.finditer(r'[A-Za-z0-9_./\-]*[A-Za-z0-9_\-]\.sql', body):
        base = os.path.basename(m.group(0))
        if base in MIGS and base not in names:
            names.append(base)
    return names

def sql_files_of_registration():
    reg = registered()
    sqls = []  # (line, file)
    pys = []
    for ln, kind, f in reg:
        (sqls if kind == 'sql' else pys).append((ln, f))
    return reg, sqls, pys
```
（コメント除去は簡易版: `--` を文字列の中でも切る。`/* */` は行数を保って消す。ドル引用の本文は残す。）
