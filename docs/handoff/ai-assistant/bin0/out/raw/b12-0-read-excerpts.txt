626:     def vacuum(self) -> int:
627:         """VACUUM to reclaim space after large deletes (SQLite never shrinks on its own).
628:         Takes an exclusive lock: callers must ensure no other writers are active. FTS5
629:         segments are merged first (``'optimize'``) so their pages are reclaimed too;
630:         returns the number of FTS indexes optimized (0 on merge failure / no FTS). A quarantined
631:         handle (corrupt image, replaced file, lost WAL generation) raises before any rewrite: a
632:         VACUUM reads every page and commits the result back, turning contained damage into an
633:         amplified one. Same guard ``_execute_write`` applies to every write."""
634:         with self._writer("vacuum") as holder:
635:             holder._raise_if_db_corrupt()
636:             optimized = 0
637:             try:
638:                 optimized = _optimize_fts(holder)  # manages its own lock
639:             except Exception as exc:  # health: allow BLE001 -- a failed FTS optimize is logged and VACUUM still runs
640:                 logger.warning("FTS optimize before VACUUM failed: %s", exc)
641:             with holder._lock:
642:                 holder._raise_if_db_replaced()
643:                 if holder._conn is None:
644:                     holder._reopen_after_close_locked(context="write")
645:                 # PASSIVE, not TRUNCATE: a manual `hermes sessions vacuum` runs in a transient CLI
646:                 # process; a TRUNCATE reset here would race a live gateway writer.
647:                 _try_checkpoint(holder, "PASSIVE", "WAL checkpoint (PASSIVE) before VACUUM failed: %s")
648:                 holder._conn.execute("VACUUM")
649:                 # VACUUM rewrites every page THROUGH the WAL; without this TRUNCATE a 3 GB DB leaves a 3 GB -wal.
650:                 _try_checkpoint(holder, "TRUNCATE", "WAL checkpoint (TRUNCATE) after VACUUM failed: %s")
651:                 # TRUNCATE may replace the WAL inode; adopt the new sidecars so the
652:                 # write-path generation guard does not halt this connection.
653:                 holder._record_db_file_identity()
654:             return optimized
