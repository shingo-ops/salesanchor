## 台本 /tmp/recon/q01.py の全文
```python
import ast, os, sys
ROOT = "/root/worktrees/salesanchor/release-deploy-stall-b-recon"
TARGETS = {"create_engine", "create_async_engine", "connect", "create_pool"}
rows = []
for d, _, fs in os.walk(os.path.join(ROOT, "backend/app")):
    for f in sorted(fs):
        if not f.endswith(".py"):
            continue
        p = os.path.join(d, f)
        rel = os.path.relpath(p, ROOT)
        src = open(p, encoding="utf-8").read()
        try:
            tree = ast.parse(src)
        except Exception as e:
            print("PARSE-FAIL", rel, e)
            continue
        for n in ast.walk(tree):
            if not isinstance(n, ast.Call):
                continue
            fn = n.func
            if isinstance(fn, ast.Attribute):
                nm = fn.attr
                base = ast.unparse(fn.value)
            elif isinstance(fn, ast.Name):
                nm = fn.id
                base = ""
            else:
                continue
            kind = None
            if nm in ("create_engine", "create_async_engine"):
                kind = nm
            elif nm == "connect" and base.endswith(("psycopg2", "asyncpg")):
                kind = base.split(".")[-1] + ".connect"
            elif nm == "create_pool" and base.endswith("asyncpg"):
                kind = "asyncpg.create_pool"
            if kind is None:
                continue
            seg = ast.get_source_segment(src, n) or ""
            how = []
            if "application_name" in seg:
                how.append("arg:application_name")
            if "server_settings" in seg:
                how.append("arg:server_settings")
            if "connect_args" in seg:
                how.append("arg:connect_args")
            if any(k.arg is None for k in n.keywords):
                how.append("**dict")
            rows.append((rel, n.lineno, kind, ",".join(how) or "-", "async" if "async" in kind or "asyncpg" in kind else "sync",
                         "file-has-application_name" if "application_name" in src else "file-no-application_name"))
rows.sort()
print("COUNT", len(rows))
for r in rows:
    print(" | ".join(map(str, r)))
# 別名: connect を変数に入れる形
print("--- alias candidates (assign of psycopg2.connect/asyncpg.connect/create_pool without call) ---")
for d, _, fs in os.walk(os.path.join(ROOT, "backend/app")):
    for f in sorted(fs):
        if f.endswith(".py"):
            p = os.path.join(d, f)
            for i, line in enumerate(open(p, encoding="utf-8"), 1):
                s = line.strip()
                if ("= psycopg2.connect" in s or "= asyncpg.connect" in s or "= asyncpg.create_pool" in s or "= create_engine" in s or "= create_async_engine" in s) and "(" not in s.split("=")[1].split("connect")[-1][:1]:
                    pass
                if ("import psycopg2" in s or "import asyncpg" in s or "from psycopg2" in s or "from asyncpg" in s):
                    print(os.path.relpath(p, ROOT), i, s)
```
