     1	"""試しの道具（読み取りだけ）が、計算や Gemini 待ちの間に DB の取引を開いたままにしないための共通の関数。
     2	
     3	設計: docs/handoff/replay-tx-release/design.md §3 C4
     4	- release_before_compute: 計算・外の呼び出しの前に取引を閉じる（読み取りだけなので rollback。取引が無ければ何もしない）
     5	- set_idle_tx_limit: 開いた取引の中で待つ時間の上限を、この接続にだけ置く（commit で確定させる）
     6	"""
     7	from __future__ import annotations
     8	
     9	import logging
    10	
    11	from sqlalchemy import event, text
    12	from sqlalchemy.engine import Engine
    13	from sqlalchemy.orm import Session
    14	
    15	logger = logging.getLogger(__name__)
    16	
    17	IDLE_TX_LIMIT_SECONDS = 30
    18	
    19	
    20	def release_before_compute(session: Session) -> None:
    21	    session.rollback()
    22	
    23	
    24	def _on_connect_set_idle_limit(dbapi_connection, connection_record) -> None:
    25	    """新しい DB 接続ができるたびに、開いた取引で待つ上限を置く（design §14 C5'）。
    26	
    27	    commit まで流す。プールが接続を返すときの rollback で SET が消えないようにするため。
    28	    """
    29	    cursor = dbapi_connection.cursor()
    30	    try:
    31	        cursor.execute(f"SET idle_in_transaction_session_timeout = '{int(IDLE_TX_LIMIT_SECONDS)}s'")
    32	    finally:
    33	        cursor.close()
    34	    dbapi_connection.commit()
    35	
    36	
    37	def install_idle_tx_listener(engine: Engine) -> None:
    38	    """engine に connect の受け手を1回だけ付ける。付いていれば何もしない。"""
    39	    if not event.contains(engine, "connect", _on_connect_set_idle_limit):
    40	        event.listen(engine, "connect", _on_connect_set_idle_limit)
    41	
    42	
    43	def set_idle_tx_limit(session: Session, seconds: int = IDLE_TX_LIMIT_SECONDS) -> str:
    44	    bind = session.get_bind()
    45	    if isinstance(bind, Engine):
    46	        install_idle_tx_listener(bind)
    47	    session.execute(text(f"SET idle_in_transaction_session_timeout = '{int(seconds)}s'"))
    48	    session.commit()
    49	    value = session.execute(text("SHOW idle_in_transaction_session_timeout")).scalar()
    50	    session.rollback()
    51	    logger.info("[tx_guard] idle_in_transaction_session_timeout=%s", value)
    52	    return str(value)
